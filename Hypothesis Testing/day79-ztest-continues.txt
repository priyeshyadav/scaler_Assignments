Q1:2.4532, Hourly wages in plant 'A' are higher than those in plant 'B'.

import numpy as np
from scipy.stats import norm, binom, poisson
plantA_Sample = 150
plantA_Mean = 2.87
plantA_std = 1.08
plantB_Sample = 200
plantB_Mean = 2.56
plantB_std = 1.28
aplha = 0.01
#Ho is hourly wages paid by both plant are equal
#Ha is hourly wages paid by A are higher than B
#Test - Ride side
#This is two sample Z test

z_score = (plantA_Mean - plantB_Mean) / np.sqrt((plantA_std**2 / plantA_Sample) + (plantB_std**2 / plantB_Sample))
print(z_score)

p_value = 1 - norm.cdf(z_score)
print(p_value)

if p_value < aplha:
  print("Reject Ho")
else:
  print("Fail to reject Ho")

'''
2.453219634102559
0.007079193063952549
Reject Ho
'''

Q2: P-value: 0.1791, There is no significant evidence that Analyst Y writes less complex code compared to Analyst X.

#This is two sample Z test
#Ho : Complexity of queries for both Analyst is same

X_sample = len(Analyst_x)

p_value = norm.cdf(Z_value)
print(p_value)

if p_value < aplha:
    print("Reject the null hypothesis")
else:
    print("Fail to reject the null hypothesis")

Q6 : P-value: 0.00278, There is a significant difference in the proportion of students selected for Chasers and Seekers positions.
import numpy as np
from scipy.stats import norm, binom, poisson
c_sample = 90
c_selected = 57
s_sample = 120
s_selected = 98
alpha = 0.10
#H0 No Diff
#H1 Diff
#test two tailed
c_hat = c_selected / c_sample
s_hat = s_selected / s_sample
p_hat = (c_selected + s_selected) / (c_sample + s_sample)
z = (c_hat - s_hat) / np.sqrt(p_hat * (1 - p_hat) * ((1 / c_sample) + (1 / s_sample)))
print(z)
p_val = 2 * (1 - norm.cdf(np.abs(z)))
print(p_val)
if p_val < alpha:
    print("Reject H0")
else:
    print("Fail to reject H0")

'''
-2.990306921349541
0.002786972588957992
Reject H0
'''
